cmake_minimum_required(VERSION 3.20)
project(PortableKitFFmpegCheck LANGUAGES C CXX)
set(PSPRECOMP_GENERATED_JOBS 2 CACHE STRING "FFmpeg build jobs")
include(../../cmake/FFmpeg.cmake)
if(NOT PORTABLEKIT_HAS_FFMPEG)
    message(FATAL_ERROR "This check requires FFmpeg")
endif()
add_executable(ffmpeg_check check.c)
set_target_properties(ffmpeg_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
portablekit_use_ffmpeg(ffmpeg_check)
enable_testing()
add_test(NAME ffmpeg_check COMMAND ffmpeg_check)
